cmake_minimum_required(VERSION 3.24)
project(ShadowNet VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
option(SHADOW_FETCH_DEPENDENCIES "Fetch pinned dependencies instead of system packages" OFF)
option(SHADOW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer (Linux)" OFF)
option(SHADOW_CLANG_TIDY "Run clang-tidy on project code" OFF)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)

add_library(shadow_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(shadow_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(SHADOW_SANITIZERS)
  if(NOT UNIX OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers are currently supported on Linux GCC/Clang only")
  endif()
  target_compile_options(shadow_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(shadow_options INTERFACE -fsanitize=address,undefined)
endif()
if(SHADOW_CLANG_TIDY)
  find_program(SHADOW_TIDY_EXECUTABLE NAMES clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${SHADOW_TIDY_EXECUTABLE}")
endif()

add_subdirectory(libs)
add_subdirectory(apps)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS shadow shadow-agent shadow-gateway RUNTIME DESTINATION bin)
install(DIRECTORY config/ DESTINATION share/shadownet/config)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/shadownet)
